import numpy as np

def heuristic2(index_item, weights, capacity, nb_voisinage, voisinage, profit):
    dimension = len(capacity)
    NBITEMS = len(profit)
    heuristics_matrix = np.zeros((nb_voisinage + 1, dimension + 1))
    
    for i in range(nb_voisinage + 1):
        if i == 0:
            item_index = index_item
        else:
            item_index = voisinage[i - 1]
        
        heuristics = profit[item_index]
        for d in range(dimension):
            heuristics_matrix[i, d] = weights[d, item_index] / capacity[d]
        heuristics_matrix[i, dimension] = heuristics
    
    return heuristics_matrix
